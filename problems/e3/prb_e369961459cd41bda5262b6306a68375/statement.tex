\section{Algebraic Stories — Hilbert functions of fat points.}\label{statement}

\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Let $t,s,m\geq1$ and $n_1,\ldots,n_t\geq1$. Work over $\mathbb C$ and put
\[Y=\mathbb P^{n_1-1}\times\cdots\times\mathbb P^{n_t-1},\qquad
S=\mathbb C[x_{i,j}:1\leq i\leq t,1\leq j\leq n_i].
\]
Give $x_{i,j}$ multidegree equal to the $i$th standard basis vector. For generic distinct points $P_1,\ldots,P_s\in Y$, let $\mathfrak p_j$ be their multihomogeneous prime ideals and set
\[I_X=\bigcap_{j=1}^s\mathfrak p_j^m,\qquad X=mP_1+\cdots+mP_s.
\]
Write $\mathbb N_0=\{0,1,\ldots\}$ and $H_X(\boldsymbol a)=\dim_{\mathbb C}(S/I_X)_{\boldsymbol a}$ for $\boldsymbol a\in\mathbb N_0^t$.
\paragraph{Statement recorded in the supplied source.}
Calculate $H_X(\boldsymbol a)$ for all the parameters above and every multidegree $\boldsymbol a$. \sref{statement}{2}
\subsection{Short English statement}
Determine every multigraded Hilbert-function value for equally thickened generic points in a product of projective spaces.
