b_k(X;\F_p)=\dim_{\F_p}H_k(X;\F_p),

\]

\[
\operatorname{tors}A=\{a\in A:\text{there exists }n\in\Z_{>0}\text{ with }na=0\}.

\[
t_k(X)=\bigl|\operatorname{tors}H_k(X;\Z)\bigr|.
