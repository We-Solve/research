\[
 \Delta_{H,A}(z)=\frac{\det(zI-H)}{\det(zI-A)}
 =1-v^*(zI-A)^{-1}v,

\[

 \operatorname{tr}(zI-H)^{-1}-\operatorname{tr}(zI-A)^{-1}
