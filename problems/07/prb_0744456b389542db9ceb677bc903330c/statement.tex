\rho\colon\pi_1(\Sigma_g)\longrightarrow\operatorname{Diff}^{\omega}_{+}(\R^2)
\]

\[
p\colon E_\rho=(\widetilde{\Sigma}_g\times\R^2)/\pi_1(\Sigma_g)\longrightarrow\Sigma_g,
\qquad
\gamma\cdot(x,v)=(\gamma x,\rho(\gamma)(v)).
\]

\[
